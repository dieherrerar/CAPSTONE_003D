1️⃣ FUNCIÓN AUXILIAR → insertar_alerta_y_notify()

Esta función es llamada por la principal para insertar en detalle_alerta y emitir el NOTIFY que llega al frontend. 
-- ========================================
-- FUNCIÓN AUXILIAR: inserta y notifica alerta
-- ========================================
CREATE OR REPLACE FUNCTION insertar_alerta_y_notify(
  p_id_alerta bigint,
  p_id_dato bigint,
  p_id_dispo bigint,
  p_valor numeric,
  p_msg text,
  p_var text,
  p_ts timestamp
) RETURNS void AS $$
DECLARE
  v_nombre_alerta text;
  v_nivel text;
  v_open_modal boolean;
  v_payload jsonb;
BEGIN
  SELECT a.nombre_alerta, t.nivel_alerta
  INTO v_nombre_alerta, v_nivel
  FROM alerta a
  JOIN tipo_alerta t USING (id_tipo_alerta)
  WHERE a.id_alerta = p_id_alerta;

  INSERT INTO detalle_alerta (id_alerta, id_dato_dispositivo, fecha_hora_alerta, valor_anomalo)
  VALUES (p_id_alerta, p_id_dato, now(), p_valor);

  v_open_modal := (v_nivel IN ('alto','extremo'));

  v_payload := jsonb_build_object(
    'alert', jsonb_build_object(
      'id_alerta', p_id_alerta,
      'nombre_alerta', v_nombre_alerta,
      'id_dato_dispositivo', p_id_dato,
      'id_dispositivo', p_id_dispo,
      'nivel_alerta', v_nivel,
      'message', p_msg,
      'valor', p_valor,
      'variable', p_var,
      'timestamp', p_ts
    ),
    'open_modal', v_open_modal
  );

  PERFORM pg_notify('new_alert', v_payload::text);
END;
$$ LANGUAGE plpgsql;
