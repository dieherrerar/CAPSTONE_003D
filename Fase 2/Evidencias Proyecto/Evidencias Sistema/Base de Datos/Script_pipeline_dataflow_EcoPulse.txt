# ==========================================
# 🌎 Dataflow: Pub/Sub → PostgreSQL (Ecopulse)
# ==========================================
import json
import psycopg2
import apache_beam as beam
from apache_beam.options.pipeline_options import PipelineOptions, StandardOptions, GoogleCloudOptions, SetupOptions

# ==========================================
# ⚙️ CONFIGURACIÓN GLOBAL
# ==========================================
PROJECT_ID = "gentle-complex-473322-d9"
REGION = "us-central1"
TOPIC_SUBSCRIPTION = f"projects/{PROJECT_ID}/subscriptions/datos_dispositivos-sub"

DB_CONFIG = {
    "dbname": "EcoPulse_Prod",
    "user": "postgres",
    "password": "Admin!123",  # ⚠️ cambia esto
    "host": "172.16.32.3",            # ⚠️ IP interna de tu base de datos en Cloud SQL
    "port": "5432"
}

# ==========================================
# 🧩 TRANSFORMACIÓN DE DATOS
# ==========================================
class TransformarDatos(beam.DoFn):
    def process(self, element):
        registro = json.loads(element)
        for k, v in registro.items():
            if v is None or v == "" or str(v).lower() == "nan":
                registro[k] = -999
        yield registro

# ==========================================
# 🧩 INSERCIÓN EN POSTGRESQL
# ==========================================
class InsertarPostgres(beam.DoFn):
    def start_bundle(self):
        try:
            self.conn = psycopg2.connect(**DB_CONFIG)
            self.cur = self.conn.cursor()
        except Exception as e:
            print("❌ Error conectando a PostgreSQL:", e)

    def process(self, element):
        try:
            sql = """
                INSERT INTO datos_dispositivo (
                    tem_bme280, hum_bme280, pres_bme280, alt_bme280,
                    "mp1.0_stp", "mp2.5_stp", mp10_stp,
                    "mp1.0_ate", "mp2.5_ate", mp10_ate,
                    "mp_gt_0.3um", "mp_gt_0.5um", "mp_gt_1.0um",
                    "mp_gt_2.5um", "mp_gt_5.0um", mp_gt_10um,
                    co2_mhz19, dir_viento, rap_viento,
                    agua_caida, consumo_1, consumo_2, consumo_3, id_dispositivo
                )
                VALUES (
                    %(tem_bme280)s, %(hum_bme280)s, %(pres_bme280)s, %(alt_bme280)s,
                    %(mp1.0_stp)s, %(mp2.5_stp)s, %(mp10_stp)s,
                    %(mp1.0_ate)s, %(mp2.5_ate)s, %(mp10_ate)s,
                    %(mp_gt_0.3um)s, %(mp_gt_0.5um)s, %(mp_gt_1.0um)s,
                    %(mp_gt_2.5um)s, %(mp_gt_5.0um)s, %(mp_gt_10um)s,
                    %(co2_mhz19)s, %(dir_viento)s, %(rap_viento)s,
                    %(agua_caida)s, %(consumo_1)s, %(consumo_2)s, %(consumo_3)s, %(id_dispositivo)s
                )
            """
            self.cur.execute(sql, element)
            self.conn.commit()
            print(f"✅ Insertado en PostgreSQL: {element}")

        except Exception as e:
            print("⚠️ Error insertando registro:", e)

    def finish_bundle(self):
        if hasattr(self, 'cur') and self.cur:
            self.cur.close()
        if hasattr(self, 'conn') and self.conn:
            self.conn.close()

# ==========================================
# 🚀 FUNCIÓN PRINCIPAL
# ==========================================
def run():
    options = PipelineOptions(
        streaming=True,
        save_main_session=True
    )

    gcp_options = options.view_as(GoogleCloudOptions)
    gcp_options.project = PROJECT_ID
    gcp_options.region = REGION
    gcp_options.job_name = "pubsub-to-postgres-transform"
    gcp_options.temp_location = f"gs://{PROJECT_ID}-temp"
    gcp_options.service_account_email = "dataflow-ecopulse@gentle-complex-473322-d9.iam.gserviceaccount.com"

    options.view_as(StandardOptions).runner = "DataflowRunner"

# ✅ Configuración específica para VPC Connector
    options._all_options["experiments"] = [
        "use_runner_v2",
        "enable_google_cloud_vpc",
        "use_dataflow_serverless_vpc_access"
    ]
    options._all_options["vpc_connector"] = (
        "projects/gentle-complex-473322-d9/locations/us-central1/connectors/dataflow-ecopulse-db"
    )
    options._all_options["worker_region"] = "us-central1"

    options.view_as(SetupOptions).save_main_session = True

    # Pipeline principal
    with beam.Pipeline(options=options) as p:
        (
            p
            | "Leer desde Pub/Sub" >> beam.io.ReadFromPubSub(subscription=TOPIC_SUBSCRIPTION).with_output_types(bytes)
            | "Decodificar JSON" >> beam.Map(lambda x: x.decode("utf-8"))
            | "Transformar datos" >> beam.ParDo(TransformarDatos())
            | "Insertar en PostgreSQL" >> beam.ParDo(InsertarPostgres())
        )

# ==========================================
# 🏁 EJECUCIÓN
# ==========================================
if __name__ == "__main__":
    run()
