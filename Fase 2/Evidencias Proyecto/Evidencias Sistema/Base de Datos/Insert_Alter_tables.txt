#captura automática del dia de la semana
-- Función para asignar el día de la semana
CREATE OR REPLACE FUNCTION set_dia_semana()
RETURNS TRIGGER AS $$
BEGIN
  NEW.dia_semana := TO_CHAR(NEW.fecha_registro, 'TMDay');
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger que ejecuta la función en cada INSERT o UPDATE
CREATE TRIGGER trg_set_dia_semana
BEFORE INSERT OR UPDATE ON datos_dispositivo
FOR EACH ROW
EXECUTE FUNCTION set_dia_semana();
