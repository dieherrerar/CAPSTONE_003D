-- ========================================
-- FUNCIÓN PRINCIPAL: procesa datos insertados
-- ========================================
CREATE OR REPLACE FUNCTION public.fn_procesar_dato_dispositivo() RETURNS trigger AS $$
DECLARE
  v_now timestamp := now();
  v_id_dispo bigint := NEW.id_dispositivo;
  v_dato_id bigint := NEW.id_dato_dispositivo;

  -- UMBRALES
  th_temp_calor_crit  numeric := 37.0;  
  th_temp_calor_warn1 numeric := 33.0;  
  th_temp_frio_warn1  numeric := -4.0;  
  th_temp_frio_crit   numeric := -5.0;  
  th_lluvia_dia_crit  numeric := 80.0;  
  th_pm_pre_emerg     numeric := 80.0;  
  th_pm_emerg         numeric := 150.0; 
  th_huracan_kmh      numeric := 119.0; 
  th_co2_delta_ppm    numeric := 150.0; 

  v_viento_kmh numeric := CASE WHEN NEW.rap_viento IS NULL THEN NULL ELSE NEW.rap_viento * 3.6 END;
  v_mean_co2 numeric;
  v_delta_co2 numeric;
  v_cols_over100 int := 0;
BEGIN
  -- ========================================
  -- 1️⃣ ERRORES DE SENSOR (ahora < -100)
  -- ========================================
  SELECT COUNT(*) INTO v_cols_over100
  FROM (
    VALUES
      (NEW.tem_bme280),(NEW.hum_bme280),(NEW.pres_bme280),(NEW.alt_bme280),
      (NEW."mp1.0_stp"),(NEW."mp2.5_stp"),(NEW.mp10_stp),
      (NEW."mp1.0_ate"),(NEW."mp2.5_ate"),(NEW.mp10_ate),
      (NEW."mp_gt_0.3um"),(NEW."mp_gt_0.5um"),(NEW."mp_gt_1.0um"),
      (NEW."mp_gt_2.5um"),(NEW."mp_gt_5.0um"),(NEW.mp_gt_10um),
      (NEW.co2_mhz19),(NEW.rap_viento),(NEW.agua_caida),
      (NEW.consumo_1),(NEW.consumo_2),(NEW.consumo_3)
  ) AS t(val)
  WHERE val IS NOT NULL AND val < -100;

  IF v_cols_over100 = 1 THEN
    -- id_alerta = 1 (error de sensor individual)
    PERFORM insertar_alerta_y_notify(
      1, v_dato_id, v_id_dispo,
      -101.0,
      'Error de lectura en un sensor (valor < -100)',
      'var_desconocida', NEW.timestamp_registro
    );
  ELSIF v_cols_over100 >= 2 THEN
    -- id_alerta = 3 (errores múltiples)
    PERFORM insertar_alerta_y_notify(
      3, v_dato_id, v_id_dispo,
      -101.0,
      'Errores de lectura en múltiples sensores (valores < -100)',
      'var_multiple', NEW.timestamp_registro
    );
  END IF;

  -- ========================================
  -- 2️⃣ TEMPERATURA (4,5,8,9)
  -- ========================================
  IF NEW.tem_bme280 IS NOT NULL THEN
    IF NEW.tem_bme280 >= th_temp_calor_crit THEN
      PERFORM insertar_alerta_y_notify(8, v_dato_id, v_id_dispo, NEW.tem_bme280,
        format('Temperatura crítica %.1f °C (≥ %.1f)', NEW.tem_bme280, th_temp_calor_crit),
        'tem_bme280', NEW.timestamp_registro);
    ELSIF NEW.tem_bme280 >= th_temp_calor_warn1 AND NEW.tem_bme280 < th_temp_calor_crit THEN
      PERFORM insertar_alerta_y_notify(5, v_dato_id, v_id_dispo, NEW.tem_bme280,
        format('Temperatura elevada %.1f °C (33–36)', NEW.tem_bme280),
        'tem_bme280', NEW.timestamp_registro);
    ELSIF NEW.tem_bme280 <= th_temp_frio_crit THEN
      PERFORM insertar_alerta_y_notify(9, v_dato_id, v_id_dispo, NEW.tem_bme280,
        format('Temperatura crítica %.1f °C (≤ %.1f)', NEW.tem_bme280, th_temp_frio_crit),
        'tem_bme280', NEW.timestamp_registro);
    ELSIF NEW.tem_bme280 <= 0 AND NEW.tem_bme280 >= th_temp_frio_warn1 THEN
      PERFORM insertar_alerta_y_notify(4, v_dato_id, v_id_dispo, NEW.tem_bme280,
        format('Temperatura baja %.1f °C (0 a -4)', NEW.tem_bme280),
        'tem_bme280', NEW.timestamp_registro);
    END IF;
  END IF;

  -- ========================================
  -- 3️⃣ MATERIAL PARTICULADO (6,13)
  -- ========================================
  IF COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp") IS NOT NULL THEN
    IF COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp") >= th_pm_emerg THEN
      PERFORM insertar_alerta_y_notify(13, v_dato_id, v_id_dispo,
        COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp"),
        format('MP2.5 EMERGENCIA (%.1f µg/m³)', COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp")),
        'pm2.5', NEW.timestamp_registro);
    ELSIF COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp") >= th_pm_pre_emerg THEN
      PERFORM insertar_alerta_y_notify(6, v_dato_id, v_id_dispo,
        COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp"),
        format('MP2.5 PRE-EMERGENCIA (%.1f µg/m³)', COALESCE(NEW."mp2.5_ate", NEW."mp2.5_stp")),
        'pm2.5', NEW.timestamp_registro);
    END IF;
  END IF;

  -- ========================================
  -- 4️⃣ CO₂ (7)
  -- ========================================
  IF NEW.co2_mhz19 IS NOT NULL THEN
    SELECT AVG(co2_mhz19) INTO v_mean_co2
    FROM datos_dispositivo
    WHERE id_dispositivo = v_id_dispo
      AND co2_mhz19 IS NOT NULL
      AND timestamp_registro >= (NEW.timestamp_registro - interval '60 minutes');

    IF v_mean_co2 IS NOT NULL THEN
      v_delta_co2 := NEW.co2_mhz19 - v_mean_co2;
      IF v_delta_co2 >= th_co2_delta_ppm THEN
        PERFORM insertar_alerta_y_notify(7, v_dato_id, v_id_dispo, NEW.co2_mhz19,
          format('CO₂ +%.0f ppm sobre media (μ=%.0f)', v_delta_co2, v_mean_co2),
          'co2_mhz19', NEW.timestamp_registro);
      END IF;
    END IF;
  END IF;

  -- ========================================
  -- 5️⃣ LLUVIA ACUMULADA (10)
  -- ========================================
  IF NEW.agua_caida IS NOT NULL THEN
    DECLARE v_acc_dia numeric;
    BEGIN
      SELECT COALESCE(SUM(agua_caida),0) INTO v_acc_dia
      FROM datos_dispositivo
      WHERE id_dispositivo = v_id_dispo
        AND fecha_registro = NEW.fecha_registro;

      IF v_acc_dia >= th_lluvia_dia_crit THEN
        PERFORM insertar_alerta_y_notify(10, v_dato_id, v_id_dispo, v_acc_dia,
          format('Lluvia acumulada diaria %.1f mm (≥ %.1f)', v_acc_dia, th_lluvia_dia_crit),
          'agua_caida', NEW.timestamp_registro);
      END IF;
    END;
  END IF;

  -- ========================================
  -- 6️⃣ HURACÁN (14)
  -- ========================================
  IF v_viento_kmh IS NOT NULL AND v_viento_kmh >= th_huracan_kmh THEN
    PERFORM insertar_alerta_y_notify(14, v_dato_id, v_id_dispo, v_viento_kmh,
      format('Viento sostenido %.1f km/h (≥ %.1f)', v_viento_kmh, th_huracan_kmh),
      'rap_viento', NEW.timestamp_registro);
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;
